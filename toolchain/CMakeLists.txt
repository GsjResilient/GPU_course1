cmake_minimum_required(VERSION 3.16)
project(NutShellToolchain VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NTSIM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../NutShellGPU_sim")
add_executable(ntsim native/ntsim.cpp "${NTSIM_DIR}/func/func_sim.cpp" "${NTSIM_DIR}/timing/cycle_sim.cpp")
target_include_directories(ntsim PRIVATE "${NTSIM_DIR}")
if(MSVC)
    target_compile_options(ntsim PRIVATE /utf-8 /fp:strict)
else()
    target_compile_options(ntsim PRIVATE -frounding-math -ffp-contract=off -fno-strict-aliasing)
endif()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    enable_testing()
    add_test(NAME toolchain COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_toolchain.py")
    set_tests_properties(toolchain PROPERTIES ENVIRONMENT "NTSIM_EXECUTABLE=$<TARGET_FILE:ntsim>")
endif()
